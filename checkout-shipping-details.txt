function getAuthToken(): string | null {

  const keys = [

    "access_token",

    "accessToken",

    "token",

    "authToken",

    "jwt",

    "fockis_token",

    "fockis_auth_token",

  ];

  for (const key of keys) {

    const value = localStorage.getItem(key);

    if (value?.trim()) {

      return value.trim();

    }

  }

  return null;

}

function buildUrl(path: string): string {

  const normalizedPath = path.startsWith("/") ? path : `/${path}`;

  return `${API_BASE_URL}${normalizedPath}`;

}

async function extractErrorMessage(response: Response): Promise<string> {

  const contentType = response.headers.get("content-type") || "";

  if (contentType.includes("application/json")) {

    try {

      const body: unknown = await response.json();

      if (body && typeof body === "object") {

        const record = body as Record<string, unknown>;

        if (typeof record.message === "string") {

          return record.message;

        }

        if (Array.isArray(record.message)) {

          return record.message.join(", ");

        }

        if (typeof record.error === "string") {

          return record.error;

        }

      }

    } catch {

      // Ignore invalid JSON.

    }

  } else {

    try {

      const text = await response.text();

      if (text) return text;

    } catch {

      // Ignore response parsing errors.

    }

  }

  return `Request failed with status ${response.status}`;

}

async function post<T>(

  path: string,

  body: unknown,

  authenticated = true,

): Promise<T> {

  const token = authenticated ? getAuthToken() : null;

  if (authenticated && !token) {

    throw new Error("UNAUTHORIZED");

  }

  const headers = new Headers({

    Accept: "application/json",

    "Content-Type": "application/json",

  });

  if (token) {

    headers.set("Authorization", `Bearer ${token}`);

  }

  const response = await fetch(buildUrl(path), {

    method: "POST",

    headers,

    credentials: "include",

    body: JSON.stringify(body),

  });

  if (!response.ok) {

    if (response.status === 401) {

      throw new Error("UNAUTHORIZED");

    }

    if (response.status === 403) {

      throw new Error("FORBIDDEN");

    }

    if (response.status === 404) {

      throw new Error("CHECKOUT_ENDPOINT_NOT_FOUND");

    }

    throw new Error(await extractErrorMessage(response));

  }

  if (response.status === 204) {

    return undefined as T;

  }

  const contentType = response.headers.get("content-type") || "";

  if (!contentType.includes("application/json")) {

    const text = await response.text();

    return text as T;

  }

  return response.json() as Promise<T>;

}

function getNestedString(

  value: unknown,

  keys: string[],

): string | undefined {

  if (!value || typeof value !== "object") {

    return undefined;

  }

  const record = value as Record<string, unknown>;

  for (const key of keys) {

    const candidate = record[key];

    if (typeof candidate === "string" && candidate.trim()) {

      return candidate.trim();

    }

  }

  return undefined;

}

function getNestedNumber(

  value: unknown,

  keys: string[],

): number | undefined {

  if (!value || typeof value !== "object") {

    return undefined;

  }

  const record = value as Record<string, unknown>;

  for (const key of keys) {

    const candidate = record[key];

    if (typeof candidate === "number" && Number.isFinite(candidate)) {

      return candidate;

    }

    if (

      typeof candidate === "string" &&

      candidate.trim() &&

      Number.isFinite(Number(candidate))

    ) {

      return Number(candidate);

    }

  }

  return undefined;

}

function getCartStoreIds(items: CartLineItem[]): string[] {

  const storeIds = new Set<string>();

  for (const item of items) {

    const storeId = String(item.storeId || "").trim();

    if (storeId) {

      storeIds.add(storeId);

    }

  }

  if (storeIds.size === 0) {

    throw new Error(

      "Your cart is missing its seller store ID. Refresh your cart and try again.",

    );

  }

  return Array.from(storeIds);

}

function parseEta(eta: string): { min: number; max: number } {

  const normalizedEta = String(eta || "").trim();

  const range = normalizedEta.match(/(\d+)\s*-\s*(\d+)/);

  if (range) {

    return {

      min: Number(range[1]),

      max: Number(range[2]),

    };

  }

  const single = normalizedEta.match(/(\d+)/);

  if (single) {

    const days = Number(single[1]);

    return { min: days, max: days };

  }

  return { min: 2, max: 4 };

}

/**

 * Creates an approximate parcel for each seller.

 *

 * Product shipping fields can be supplied on cart items as:

 * shippingWeight, shippingLength, shippingWidth, shippingHeight.

 * Generic weight/length/width/height fields are also accepted.

 *

 * Values are assumed to be pounds and inches. Confirm the units

 * used by your product form before enabling live rates.

 */

/**
 * Builds a Shippo parcel for one seller's cart items.
 *
 * Preferred product schema:
 *   weightKg: kilograms
 *   dimensionsCm: { length, width, height } in centimeters
 *
 * Legacy shippingWeight/weight and shippingLength/length/etc. are also
 * supported and are interpreted as pounds and inches.
 */
function getStoreParcel(items: CartLineItem[]): ShippoParcel {
  const readPathNumber = (
    source: unknown,
    path: string[],
  ): number | undefined => {
    let current: unknown = source;

    for (const key of path) {
      if (!current || typeof current !== "object") {
        return undefined;
      }
      current = (current as Record<string, unknown>)[key];
    }

    if (typeof current === "number" && Number.isFinite(current)) {
      return current;
    }

    if (typeof current === "string" && current.trim() !== "") {
      const parsed = Number(current);
      if (Number.isFinite(parsed)) return parsed;
    }

    return undefined;
  };

  const firstNumber = (
    source: unknown,
    paths: string[][],
  ): number | undefined => {
    for (const path of paths) {
      const value = readPathNumber(source, path);
      if (value !== undefined) return value;
    }
    return undefined;
  };

  let weight = 0;
  let length = 0;
  let width = 0;
  let height = 0;

  for (const item of items) {
    const record = item as CartLineItem & Record<string, unknown>;
    const quantity = Math.max(
      1,
      Math.floor(readPathNumber(record, ["quantity"]) ?? 1),
    );

    const metricWeightKg = firstNumber(record, [
      ["weightKg"],
      ["shipping", "weightKg"],
      ["product", "weightKg"],
      ["product", "shipping", "weightKg"],
    ]);
    const metricLengthCm = firstNumber(record, [
      ["dimensionsCm", "length"],
      ["shipping", "dimensionsCm", "length"],
      ["product", "dimensionsCm", "length"],
      ["product", "shipping", "dimensionsCm", "length"],
    ]);
    const metricWidthCm = firstNumber(record, [
      ["dimensionsCm", "width"],
      ["shipping", "dimensionsCm", "width"],
      ["product", "dimensionsCm", "width"],
      ["product", "shipping", "dimensionsCm", "width"],
    ]);
    const metricHeightCm = firstNumber(record, [
      ["dimensionsCm", "height"],
      ["shipping", "dimensionsCm", "height"],
      ["product", "dimensionsCm", "height"],
      ["product", "shipping", "dimensionsCm", "height"],
    ]);

    // Product metric fields are converted to Shippo's lb/in units.
    const itemWeight =
      metricWeightKg !== undefined
        ? metricWeightKg * 2.2046226218
        : firstNumber(record, [
            ["shippingWeight"],
            ["weight"],
            ["productWeight"],
            ["product", "shippingWeight"],
            ["product", "weight"],
            ["product", "productWeight"],
          ]);

    const itemLength =
      metricLengthCm !== undefined
        ? metricLengthCm / 2.54
        : firstNumber(record, [
            ["shippingLength"],
            ["length"],
            ["product", "shippingLength"],
            ["product", "length"],
          ]);

    const itemWidth =
      metricWidthCm !== undefined
        ? metricWidthCm / 2.54
        : firstNumber(record, [
            ["shippingWidth"],
            ["width"],
            ["product", "shippingWidth"],
            ["product", "width"],
          ]);

    const itemHeight =
      metricHeightCm !== undefined
        ? metricHeightCm / 2.54
        : firstNumber(record, [
            ["shippingHeight"],
            ["height"],
            ["product", "shippingHeight"],
            ["product", "height"],
          ]);

    if (
      itemWeight === undefined ||
      itemWeight <= 0 ||
      itemLength === undefined ||
      itemLength <= 0 ||
      itemWidth === undefined ||
      itemWidth <= 0 ||
      itemHeight === undefined ||
      itemHeight <= 0
    ) {
      const name =
        typeof record.productName === "string"
          ? record.productName
          : "a product";

      throw new Error(
        `Shipping weight and dimensions are missing for ${name}. ` +
          "Add valid product package weight and dimensions before calculating live rates.",
      );
    }

    weight += itemWeight * quantity;
    length = Math.max(length, itemLength);
    width = Math.max(width, itemWidth);
    height += itemHeight * quantity;
  }

  if (
    !Number.isFinite(weight) ||
    !Number.isFinite(length) ||
    !Number.isFinite(width) ||
    !Number.isFinite(height) ||
    weight <= 0 ||
    length <= 0 ||
    width <= 0 ||
    height <= 0
  ) {
    throw new Error("Valid shipping package dimensions are required.");
  }

  return {
    weight: Number(weight.toFixed(4)),
    massUnit: "lb",
    length: Number(length.toFixed(4)),
    width: Number(width.toFixed(4)),
    height: Number(height.toFixed(4)),
    distanceUnit: "in",
  };
}

function normalizeShippoRate(

  rate: ShippoCheckoutRate,

  storeId: string,

): ShippingMethodOption {

  const days =

    typeof rate.estimatedDays === "number" &&

    Number.isFinite(rate.estimatedDays) &&

    rate.estimatedDays > 0

      ? rate.estimatedDays

      : null;

  const etaLabel = days

    ? `${days} day${days === 1 ? "" : "s"}`

    : "delivery time varies";

  return {

    id: rate.rateId,

    storeId,

    type: "national_delivery",

    label: [rate.carrier, rate.service, etaLabel]

      .filter(Boolean)

      .join(" · "),

    price: rate.amount,

    estimatedDays: {

      min: days ?? 0,

      max: days ?? 0,

    },

  };

}

/**

 * Requests real Shippo checkout rates, separately for each seller.

 * The backend must implement POST /fockis-shop/shipping/shippo/checkout-rates.

 */

export async function getShippingEstimates(

  items: CartLineItem[],

  destination?: ShippoCheckoutDestination,

): Promise<ShippingEstimate[]> {

  if (!items.length) {

    return [];

  }

  if (!destination) {

    throw new Error(

      "Enter your delivery address to calculate live shipping rates.",

    );

  }

  const requiredAddress = [

    destination.fullName,

    destination.street,

    destination.city,

    destination.state,

    destination.zipCode,

    destination.country,

    destination.email,

  ];

  if (requiredAddress.some((value) => !String(value ?? "").trim())) {

    throw new Error(

      "Complete your name, street address, city, state, postal code, country, and email to calculate shipping.",

    );

  }

  const storeIds = getCartStoreIds(items);

  if (

    storeIds.some(

      (storeId) =>

        storeId === "fockis-shop" ||

        !/^[a-f\d]{24}$/i.test(storeId),

    )

  ) {

    throw new Error(

      "A valid seller store ID is missing from your cart. Refresh your cart and try again.",

    );

  }

  const estimates = await Promise.all(

    storeIds.map(async (storeId): Promise<ShippingEstimate> => {

      const storeItems = items.filter(

        (item) => String(item.storeId || "").trim() === storeId,

      );

      const parcel = getStoreParcel(storeItems);

      const response = await post<ShippoCheckoutRatesResponse>(

        "/fockis-shop/shipping/shippo/checkout-rates",

        {

          storeId,

          destination,

          parcel,

        },

        true,

      );

      const options = (response.rates ?? [])

        .filter(

          (rate) =>

            Boolean(rate.rateId) &&

            Number.isFinite(rate.amount) &&

            rate.amount >= 0,

        )

        .map((rate) => normalizeShippoRate(rate, storeId));

      if (!options.length) {

        throw new Error(

          `Shippo returned no shipping options for seller ${storeId}. Check the seller's shipping origin and the delivery address.`,

        );

      }

      return {

        storeId,

        options,

        selectedOptionId: options[0].id,

      };

    }),

  );

  return estimates;

}

export async function placeOrder(

  requestData: PlaceOrderRequest,

): Promise<Order> {

  if (!requestData.items.length) {

    throw new Error("CART_EMPTY");

  }

  if (

    !Number.isFinite(requestData.subtotal) ||

    requestData.subtotal < 0

  ) {

    throw new Error("INVALID_SUBTOTAL");

  }

  if (

    !Number.isFinite(requestData.shippingCost) ||

    requestData.shippingCost < 0

  ) {

    throw new Error("INVALID_SHIPPING_COST");

  }

  if (!Number.isFinite(requestData.tax) || requestData.tax < 0) {

    throw new Error("INVALID_TAX");

  }

  if (

    !Number.isFinite(requestData.totalAmount) ||

    requestData.totalAmount < 0

  ) {

    throw new Error("INVALID_TOTAL");

  }

  if (

    requestData.paymentMethod === "card" &&

    !requestData.paymentIntentId

  ) {

    throw new Error("PAYMENT_INTENT_REQUIRED");

  }

  return post<Order>(

    "/fockis-shop/orders",

    requestData,

    true,

  );

}

export const checkoutApi = {

  getShippingEstimates,

  placeOrder,

};

export default checkoutApi;
