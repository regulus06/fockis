import React, { useEffect, useState } from "react";
import { useNavigate } from "react-router-dom";

import useSubscription from "../hooks/useSubscription";
import {
  createCheckout,
  getSubscriptionPlans,
} from "../services/subscriptionApi";

import type {
  BillingInterval,
  SubscriptionPlan,
  SubscriptionPlanConfig,
} from "../types/subscriptionTypes";

import SubscriptionPlans from "../components/SubscriptionPlans";
import "../styles/SubscriptionPlans.scss";

export default function SubscriptionPlansPage() {
  const navigate = useNavigate();

  const {
    plan,
    loading,
    actionLoading,
    error,
  } = useSubscription();

  const [billingInterval, setBillingInterval] =
    useState<BillingInterval>("MONTHLY");

  const [plans, setPlans] =
    useState<SubscriptionPlanConfig[]>([]);

  const [plansLoading, setPlansLoading] =
    useState(true);

  const [plansError, setPlansError] =
    useState<string | null>(null);

  const [checkoutLoading, setCheckoutLoading] =
    useState(false);

  const [checkoutError, setCheckoutError] =
    useState<string | null>(null);

  useEffect(() => {
    let mounted = true;

    async function loadPlans() {
      setPlansLoading(true);
      setPlansError(null);

      try {
        const result = await getSubscriptionPlans();

        if (!mounted) return;

        setPlans(
          result.filter((item) => item.active),
        );
      } catch (err) {
        if (!mounted) return;

        setPlansError(
          err instanceof Error
            ? err.message
            : "Unable to load current subscription prices.",
        );
      } finally {
        if (mounted) {
          setPlansLoading(false);
        }
      }
    }

    void loadPlans();

    return () => {
      mounted = false;
    };
  }, []);

  const currentPlan: SubscriptionPlan =
    plan ?? "BASIC";

  const isProcessing =
    loading || actionLoading || checkoutLoading;

  const handleSelect = async (
    selectedPlan: SubscriptionPlan,
  ): Promise<void> => {
    if (selectedPlan === "BASIC") {
      navigate("/subscriptions");
      return;
    }

    if (checkoutLoading || loading || actionLoading) {
      return;
    }

    const selectedConfig = plans.find(
      (item) => item.plan === selectedPlan,
    );

    if (!selectedConfig || !selectedConfig.active) {
      setCheckoutError(
        "This subscription plan is currently unavailable.",
      );
      return;
    }

    const selectedPrice =
      billingInterval === "YEARLY"
        ? selectedConfig.yearlyPrice
        : selectedConfig.monthlyPrice;

    if (
      !Number.isFinite(selectedPrice) ||
      selectedPrice <= 0
    ) {
      setCheckoutError(
        "This plan does not have a valid price for the selected billing interval. Please contact support.",
      );
      return;
    }

    try {
      setCheckoutLoading(true);
      setCheckoutError(null);

      const result = await createCheckout({
        plan: selectedPlan as Exclude<
          SubscriptionPlan,
          "BASIC"
        >,
        billingInterval,
      });

      if (!result.checkoutUrl) {
        throw new Error(
          result.message ??
            "Stripe did not return a checkout URL.",
        );
      }

      const checkoutUrl = new URL(result.checkoutUrl);

      if (
        checkoutUrl.protocol !== "https:" ||
        !(
          checkoutUrl.hostname === "checkout.stripe.com" ||
          checkoutUrl.hostname.endsWith(".checkout.stripe.com")
        )
      ) {
        throw new Error(
          "The checkout URL is not a recognized secure Stripe Checkout URL.",
        );
      }

      window.location.assign(checkoutUrl.toString());
    } catch (err) {
      setCheckoutError(
        err instanceof Error
          ? err.message
          : "Unable to start subscription checkout.",
      );

      setCheckoutLoading(false);
    }
  };

  if (loading || plansLoading) {
    return (
      <main className="fk-subscription-plans-page">
        <div
          className="fk-subscription-loading"
          role="status"
          aria-live="polite"
        >
          Loading your Fockis account plans...
        </div>
      </main>
    );
  }

  return (
    <main className="fk-subscription-plans-page">
      <header className="fk-subscription-plans-page__header">
        <span className="fk-subscriptions-eyebrow">
          FOCKIS ACCOUNT
        </span>

        <h1>Choose Your Fockis Account</h1>

        <p>
          Find the account level that fits your needs.
          Compare features, choose your billing schedule,
          and upgrade when you are ready.
        </p>
      </header>

      <section
        className="fk-subscription-current-summary"
        aria-label="Current subscription"
      >
        <div>
          <span>YOUR CURRENT ACCOUNT</span>
          <p>
            Your current Fockis subscription level
          </p>
        </div>

        <strong>{currentPlan}</strong>
      </section>

      {(checkoutError || plansError || error) && (
        <div className="fk-subscription-error" role="alert">
          {checkoutError || plansError || error}
        </div>
      )}

      {!plansError && plans.length === 0 && (
        <div className="fk-subscription-error" role="alert">
          No active subscription plans are currently available.
          Please try again later.
        </div>
      )}

      {plansError && (
        <button
          type="button"
          className="fk-subscription-retry"
          onClick={() => window.location.reload()}
        >
          Retry loading plans
        </button>
      )}

      {!plansError && plans.length > 0 && (
        <>
          <section
            className="fk-subscription-billing-control"
            aria-labelledby="fk-billing-heading"
          >
            <div className="fk-subscription-billing-control__copy">
              <h2 id="fk-billing-heading">
                Choose your billing
              </h2>

              <p>
                Select how often you want your subscription
                to renew.
              </p>
            </div>

            <div
              className="fk-billing-selector"
              role="group"
              aria-label="Billing interval"
            >
              <button
                type="button"
                className={
                  billingInterval === "MONTHLY"
                    ? "active"
                    : ""
                }
                aria-pressed={billingInterval === "MONTHLY"}
                disabled={checkoutLoading}
                onClick={() => {
                  setBillingInterval("MONTHLY");
                  setCheckoutError(null);
                }}
              >
                Monthly
              </button>

              <button
                type="button"
                className={
                  billingInterval === "YEARLY"
                    ? "active"
                    : ""
                }
                aria-pressed={billingInterval === "YEARLY"}
                disabled={checkoutLoading}
                onClick={() => {
                  setBillingInterval("YEARLY");
                  setCheckoutError(null);
                }}
              >
                Yearly
              </button>
            </div>

            <p className="fk-subscription-billing-note">
              {billingInterval === "YEARLY"
                ? "Yearly billing is charged once per year."
                : "Monthly billing is charged once per month."}
            </p>
          </section>

          {checkoutLoading && (
            <div
              className="fk-subscription-checkout-message"
              role="status"
              aria-live="polite"
            >
              Preparing your secure Stripe checkout...
            </div>
          )}

          <SubscriptionPlans
            plans={plans}
            currentPlan={currentPlan}
            billingInterval={billingInterval}
            loading={isProcessing}
            onSelect={handleSelect}
          />

          <footer className="fk-subscription-plans-page__footer">
            <span>
              Secure payments powered by Stripe.
            </span>
            <span>
              Your selected billing interval will be used at checkout.
            </span>
          </footer>
        </>
      )}
    </main>
  );
}

/* ============================================================================
   GET PLAN CONFIGURATION
============================================================================ */

export function getSubscriptionPlanConfig(
  plan: SubscriptionPlan,
): SubscriptionPlanConfig {
  return SUBSCRIPTION_PLANS[plan];
}


/* ============================================================================
   GET PLAN RANK
============================================================================ */

export function getSubscriptionPlanRank(
  plan: SubscriptionPlan,
): number {
  return SUBSCRIPTION_PLAN_RANK[plan];
}


/* ============================================================================
   CHECK PLAN ACCESS
============================================================================ */

export function hasSubscriptionPlanAccess(
  currentPlan: SubscriptionPlan,
  requiredPlan: SubscriptionPlan,
): boolean {
  return (
    getSubscriptionPlanRank(currentPlan) >=
    getSubscriptionPlanRank(requiredPlan)
  );
}


/* ============================================================================
   CHECK PAID PLAN
============================================================================ */

export function isPaidSubscriptionPlan(
  plan: SubscriptionPlan,
): boolean {
  return plan !== "BASIC";
}


/* ============================================================================
   FORMAT PRICE
============================================================================ */

export function formatSubscriptionPrice(
  price: number,
  currency = "USD",
): string {
  return new Intl.NumberFormat(
    "en-US",
    {
      style: "currency",

      currency,

      minimumFractionDigits: 0,

      maximumFractionDigits: 2,
    },
  ).format(price);
}


/* ============================================================================
   GET BILLING PRICE
============================================================================ */

export function getSubscriptionPrice(
  config: SubscriptionPlanConfig,
  billingInterval: BillingInterval,
): number {
  return billingInterval === "YEARLY"
    ? config.yearlyPrice
    : config.monthlyPrice;
}
